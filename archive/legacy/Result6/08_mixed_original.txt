###################### 多发性硬化症单细胞进行 DSE 虚拟药物筛选
############ R语言提取基质细胞中 DSE-High vs DSE-Low 的全局差异表达谱
###### 设置工作环境
setwd("${LEGACY_STUDY_ROOT}/Result6")

###### 加载R包
library(Seurat)

###### 读取多发性硬化症单细胞全景图谱数据
print("🚀 正在载入中枢神经系统单细胞表达谱大数据集...")
GSE279180_sn_atlas <- readRDS("${LEGACY_STUDY_ROOT}/Data/SC_seq/GSE279180_sn_atlas.rds")

### 🎯【核心增加】兼容性对齐身份：洗涤并纯化出基质细胞 (SC) 亚群进行靶向计算
if ("celltype" %in% colnames(GSE279180_sn_atlas@meta.data)) {
  Idents(GSE279180_sn_atlas) <- "celltype"
} else {
  Idents(GSE279180_sn_atlas) <- "cell_type"
}
endo_cells <- subset(GSE279180_sn_atlas, idents = "SC")

# 自动锁定结果 1 的关键 Assay 槽位 (优先 originalexp)
assay_name <- ifelse("originalexp" %in% Assays(endo_cells), "originalexp", "RNA")
DefaultAssay(endo_cells) <- assay_name

### 以 DSE 的中位表达量，将基质细胞分为 High 和 Low 组
med_val <- median(endo_cells@assays[[assay_name]]@data["DSE", ])

# 容错机制：如果中位数为 0（单细胞零膨胀常见），则改用均值切分，防止高低两组细胞数极端失衡
if(med_val == 0) { med_val <- mean(endo_cells@assays[[assay_name]]@data["DSE", ]) }

endo_cells$DSE_status <- ifelse(endo_cells@assays[[assay_name]]@data["DSE", ] > med_val, "High", "Low")

### 设置聚类标识为 DSE_status
Idents(endo_cells) <- "DSE_status"

### 计算 High 组对比 Low 组的差异表达
### 💡 提示：阈值设为 0，因为神经网络需要看全景基因的变化趋势，而不仅是显著基因
print("🚀 正在计算基质细胞中 DSE 暴走表型的全景神经网络特征签名...")
therapeutic_sig <- FindMarkers(endo_cells, ident.1 = "High", ident.2 = "Low",
                               logfc.threshold = 0, min.pct = 0, assay = assay_name)

### 导出为 CSV 文件供 Python 读取
write.csv(therapeutic_sig, "DSE_therapeutic_signature.csv", row.names = TRUE)
cat("✅ 逆向基质特征签名已成功导出为: DSE_therapeutic_signature.csv\n")
############ Python 中运行 DrugReflector 虚拟筛选
######加载库
import os
# 切换至结果6工作目录
os.chdir("${LEGACY_STUDY_ROOT}/Result6/")
import pandas as pd
import drugreflector as dr

######加载并整理 R 语言导出的治疗签名
df = pd.read_csv("DSE_therapeutic_signature.csv", index_col=0)

######DrugReflector 需要一个 Pandas Series 作为输入
#使用 avg_log2FC 作为基因的特征分 (V-score 替代物)
signature = df['avg_log2FC']
#给这个转换命名，输出结果时会用到这个名字
signature.name = 'Stromal:DSE_Low->DSE_High'

######初始化 DrugReflector 深度学习模型
# 🎯【绝对锁定】指向您 Linux 服务器上真实的深度学习权重核心路径
model_paths = [
    '${LEGACY_STUDY_ROOT}/Result6/checkpoints/model_fold_0.pt',
    '${LEGACY_STUDY_ROOT}/Result6/checkpoints/model_fold_1.pt',
    '${LEGACY_STUDY_ROOT}/Result6/checkpoints/model_fold_2.pt'
]

######实例化模型 (它会自动检测你的服务器是否有 GPU/CUDA 并加速)
model = dr.DrugReflector(checkpoint_paths=model_paths)

######执行极速虚拟药物筛选
#传入我们的特征签名，让模型去寻找能诱导这种转变的药物
predictions = model.predict(signature, n_top=50)

######查看并保存绝杀结果！
#提取概率最高 (最有可能促使 DSE 表达上调) 的前 10 种化合物
prob_col = ('prob', signature.name)
print("\n🌟 极具潜力的前 10 种靶向/逆转药物 (按神经网络概率排序):")
print(predictions[prob_col].nlargest(10))
# 将预测结果完整保存为 CSV，用作文章附表
predictions.to_csv("DrugReflector_DSE_Candidates_Raw.csv")
######读取LINCS官方文件对药物进行匹配
# 1. 准备你的前 50 名预测结果，并强制提取 13 位核心库编号
prob_col = ('prob', signature.name)
top_predictions = predictions[prob_col].nlargest(50).reset_index()
top_predictions.columns = ['pert_id', 'Probability']
top_predictions['pert_id'] = top_predictions['pert_id'].astype(str).str[:13]

# 2. 读取你最新下载的 compoundinfo 字典
print("正在加载最新版 LINCS 药物字典并执行 13 位核心化清洗...")
drug_info = pd.read_csv("${LEGACY_STUDY_ROOT}/Result6/compoundinfo_beta.txt", sep='\t')

# 🎯【终极修复盾】必须把官方大字典里的 pert_id 也强行截断为 13 位！熔断因批号后缀不同导致的错位！
drug_info['pert_id'] = drug_info['pert_id'].astype(str).str[:13]

# 提取关键列
cols_to_keep = ['pert_id', 'cmap_name', 'target', 'moa', 'canonical_smiles']
cols_to_keep = [c for c in cols_to_keep if c in drug_info.columns]

# 🎯 此时基于已经清洗干净的 13 位核心编号进行去重，确保合并时绝对一对一
drug_subset = drug_info[cols_to_keep].drop_duplicates(subset=['pert_id'])

# 3. 终极精准合并！
final_annotated_results = pd.merge(top_predictions, drug_subset, on='pert_id', how='left')

print("\n🌟 全新字典精细解码大获全胜！排名前 10 的潜力药物真实身份如下：")
print(final_annotated_results.head(10))

# 导出最终的绝杀附表
final_annotated_results.to_csv("Fig6E_Final_Decoded_DSE_Targeted_Drugs.csv", index=False)
print("\n✅ 带有确切靶点与机制的终极表格已保存为 Fig6E_Final_Decoded_DSE_Targeted_Drugs.csv")
